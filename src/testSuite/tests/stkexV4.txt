;*************************************************************
;*************************************************************
;**                                                         **
;**                 tzyx <> v4 conversions                  **
;**                                                         **
;** FARG selects the item: 1966 tzyx->v4, 1967 v4->tzyx,    **
;** 1968 the bidirectional exchange.                        **
;**                                                         **
;*************************************************************
;*************************************************************
In: FL_SPCRES=0 FL_CPXRES=0 SD=0 RMODE=0 IM=2compl SS=4 WS=64
Func: fnExchangeStkToMx

;=======================================================================================================================
; tzyx->v4: T Z Y X pack into a 1x4 row vector, T first (the SLVC/SLVP coefficient order)
;=======================================================================================================================
In:  FARG=1966 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RT=LonI:"1" RZ=LonI:"-6" RY=LonI:"11" RX=LonI:"-6"
Out: EC=0 FL_CPXRES=0 FL_ASLIFT=1 RL=LonI:"-6" RT=LonI:"1" RZ=LonI:"1" RY=LonI:"1" RX=ReMa:"M1,4[1,-6,11,-6]"

; a complex coefficient anywhere turns the vector complex
In:  FARG=1966 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RT=Cplx:"0 i 1" RZ=LonI:"-6" RY=Real:"11.0" RX=LonI:"-6"
Out: EC=0 FL_ASLIFT=1 RL=LonI:"-6" RX=CxMa:"M1,4[0i1,-6i0,11i0,-6i0]"

; a string refuses before anything moves
In:  FARG=1966 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RT=LonI:"1" RZ=Stri:"This is a string" RY=LonI:"11" RX=LonI:"-6"
Out: EC=24 FL_CPXRES=0 FL_ASLIFT=0 RL=LonI:"852" RT=LonI:"1" RZ=Stri:"This is a string" RY=LonI:"11" RX=LonI:"-6"

;=======================================================================================================================
; v4->tzyx: a 4-element vector in X, row or column, unpacks to T Z Y X
;=======================================================================================================================
In:  FARG=1967 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RX=ReMa:"M1,4[1,-6,11,-6]"
Out: EC=0 FL_CPXRES=0 FL_ASLIFT=1 RL=ReMa:"M1,4[1,-6,11,-6]" RT=Real:"1" RZ=Real:"-6" RY=Real:"11" RX=Real:"-6"

; the same from a column vector
In:  FARG=1967 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RX=ReMa:"M4,1[1,-6,11,-6]"
Out: EC=0 FL_CPXRES=0 FL_ASLIFT=1 RL=ReMa:"M4,1[1,-6,11,-6]" RT=Real:"1" RZ=Real:"-6" RY=Real:"11" RX=Real:"-6"

; a complex vector unpacks to complex scalars
In:  FARG=1967 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RX=CxMa:"M1,4[0i1,-6i0,11i0,-6i2]"
Out: EC=0 FL_ASLIFT=1 RL=CxMa:"M1,4[0i1,-6i0,11i0,-6i2]" RT=Cplx:"0 i 1" RZ=Cplx:"-6 i 0" RY=Cplx:"11 i 0" RX=Cplx:"-6 i 2"

; a matrix that is not a 4-element vector refuses
In:  FARG=1967 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RT=LonI:"4" RZ=LonI:"3" RY=LonI:"2" RX=ReMa:"M2,2[1,2,3,4]"
Out: EC=21 FL_CPXRES=0 FL_ASLIFT=0 RL=LonI:"852" RT=LonI:"4" RZ=LonI:"3" RY=LonI:"2" RX=ReMa:"M2,2[1,2,3,4]"

In:  FARG=1967 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RX=ReMa:"M1,3[1,2,3]"
Out: EC=21 FL_CPXRES=0 FL_ASLIFT=0 RL=LonI:"852" RX=ReMa:"M1,3[1,2,3]"

; a scalar in X refuses with the type error
In:  FARG=1967 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RX=LonI:"5"
Out: EC=24 FL_CPXRES=0 FL_ASLIFT=0 RL=LonI:"852" RX=LonI:"5"

;=======================================================================================================================
; tzyx<>v4: a matrix in X unpacks, scalars pack
;=======================================================================================================================
In:  FARG=1968 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RT=LonI:"1" RZ=LonI:"-6" RY=LonI:"11" RX=LonI:"-6"
Out: EC=0 FL_CPXRES=0 FL_ASLIFT=1 RL=LonI:"-6" RX=ReMa:"M1,4[1,-6,11,-6]"

In:  FARG=1968 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RX=ReMa:"M1,4[1,-6,11,-6]"
Out: EC=0 FL_CPXRES=0 FL_ASLIFT=1 RL=ReMa:"M1,4[1,-6,11,-6]" RT=Real:"1" RZ=Real:"-6" RY=Real:"11" RX=Real:"-6"

; the wrong size matrix refuses instead of packing
In:  FARG=1968 FL_ASLIFT=0 FL_CPXRES=0 RL=LonI:"852" RX=ReMa:"M1,3[1,2,3]"
Out: EC=21 FL_CPXRES=0 FL_ASLIFT=0 RL=LonI:"852" RX=ReMa:"M1,3[1,2,3]"
