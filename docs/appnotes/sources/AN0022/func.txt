# TEST nesting for f(x) = x^2 - p*x - 2
LBL 'FX'                   ; f(x) = x^2 - p*x - 2  (x is the variable; p an incoming parameter)
  MVAR 'x'                 ; set up the variable if missing, and give a manual solve opportunity
  MVAR 'p'                 ; set up the variable if missing, and give a manual solve opportunity
  RCL 'x'
  ENTER ×                  ; x^2
  RCL 'x' RCL 'p' × -      ; - p*x
  2 -                      ; - 2
RTN


# TEST PLOT(SOLVE): plot the root locus x*(p) over p; expect a curve
LBL 'RT'                   ; for fixed p: solve f(x)=0, return the root x
  PGMSLV 'FX'
  0 8                      ; initial guesses: lo -> Y, hi -> X
  SOLVE 'x'                ; root left in X
RTN

LBL 'PLTROOT'              ; outer: plot the locus over p
  PGMPLT 'RT'
  -5  5                    ; plot range for p
  'PLOT(SOLVE): root locus' STO A DROPX
  SNAP PAUSE 12            ; snapshot the screen; the PAUSE below waits 1.2 s so the auto SNAP filename advances
  PLTf 'p'
  SNAP PAUSE 12            ; snapshot the screen; the PAUSE below waits 1.2 s so the auto SNAP filename advances
RTN

# TEST INT(INT) double integral: XEQ 'DBLINT' -> expect 4/3 = 1.33333...
# DBLINT = INT(0..2) [INT(0..x) t dt] dx = INT(0..2) (x^2/2) dx = 4/3   (outer over x, inner over t)
LBL 'HT'                   ; innermost integrand: h(t) = t
  MVAR 't'
  RCL 't'
RTN

  MVAR 'x'

  ∫^𝑥_𝑦 't'                ; INT(0..x) h dt left in X
