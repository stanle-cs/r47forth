# TEST PLOT(f'), one to three engines deep on one cubic. The four plots drop a degree each time: CU is x^3 - 3x^2 a cubic, D1 is 3v^2 - 6v a parabola, D2 is
# 6u - 6 a straight line crossing the axis at 1, D3 is 6 a flat line. The 3x^2 term is there to put that crossing somewhere other than the origin.
NOP ; PGMDRV and f' below are patched into the .p47, see the note at LBL 'DR'
LBL 'CU'                   ; c(x) = x^3 - 3x^2
  MVAR 'x'
  RCL 'x'
  ENTER × RCL 'x' ×        ; x^3
  RCL 'x' ENTER × 3 × -    ; - 3x^2
RTN

LBL 'D1'                   ; d/dx of CU, taken at x = v; 3v^2 - 6v
  MVAR 'v'
  PGMDRV 'CU'
  RCL 'v'                  ; the point -> X
  f' 'x'
RTN

LBL 'D2'                   ; d/dv of D1, taken at v = u; 6u - 6, two engines nested

  PGMDRV 'D1'
  RCL 'u'                  ; the point -> X

RTN

LBL 'D3'                   ; d/du of D2, taken at u = w; 6, three engines nested

  PGMDRV 'D2'
